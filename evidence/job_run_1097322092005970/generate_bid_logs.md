# 01 - Generate raw auction logs (synthetic)

Writes one gzip JSON-lines file per hour into the landing volume, mimicking the log rotation of
Halyard's bidder fleet. Re-running only writes days that are missing, so the job can run daily
and Auto Loader picks up just the new files.

All data is synthetic, generated by `src/halyard_sim.py`. No customer data is used.

### Cell 3
```python
import datetime as dt
import os
import sys

import pandas as pd

sys.path.append(os.path.abspath(".."))
import halyard_sim as hs

landing = f"/Volumes/{catalog}/{schema}/raw_landing"
out_dir = f"{landing}/auctions"
publishers = pd.read_csv(f"{landing}/reference/publishers.csv")
os.makedirs(out_dir, exist_ok=True)
existing = set(os.listdir(out_dir))

today = dt.date.today()
written, skipped, n_records = 0, 0, 0
for k in range(days, 0, -1):
    day = today - dt.timedelta(days=k)
    if any(f.startswith(f"auctions_{day:%Y-%m-%d}") for f in existing):
        skipped += 1
        continue
    recs = hs.simulate_day(day, auctions_per_day, publishers, seed=int(day.strftime("%Y%m%d")))
    files = hs.write_day(recs, out_dir, day)
    written += 1
    n_records += len(recs)
    print(f"{day}: {len(recs):,} auction records -> {len(files)} hourly files")

print(f"\nDays written: {written}, already present: {skipped}, new auction records: {n_records:,}")
print(f"Files in landing: {len(os.listdir(out_dir))}")
```
